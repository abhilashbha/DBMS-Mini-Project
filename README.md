 DBMS-Mini-Project
 Table Structure 
1. Users: Stores personal details (user_id, full_name, email, phone). 
2. Wallets: Stores the balance (wallet_id, user_id, balance). 
3. Transactions: Stores transfer records (transaction_id, sender_wallet_id, receiver_wallet_id, 
amount). 
4. AuditLog: Stores automated logs generated by triggers.

CREATE TABLE Users ( user_id INT AUTO_INCREMENT PRIMARY KEY,  
full_name VARCHAR(100) NOT NULL,  
email VARCHAR(100) UNIQUE NOT NULL,  
password_hash VARCHAR(255) NOT NULL, phone_number VARCHAR(15),  
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP );

CREATE TABLE Wallets ( wallet_id INT AUTO_INCREMENT PRIMARY KEY,  
user_id INT UNIQUE NOT NULL, balance DECIMAL(10, 2) DEFAULT 0.00,  
created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, 
FOREIGN KEY (user_id) REFERENCES Users(user_id) ON DELETE CASCADE ); 

CREATE TABLE Transactions ( transaction_id INT AUTO_INCREMENT PRIMARY KEY,  
sender_wallet_id INT,  
receiver_wallet_id INT,  
amount DECIMAL(10, 2) NOT NULL,  
transaction_type ENUM('DEPOSIT', 'WITHDRAWAL', 'TRANSFER') NOT NULL,  
transaction_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP,  
FOREIGN KEY (sender_wallet_id) REFERENCES Wallets(wallet_id) ON DELETE CASCADE,  
FOREIGN KEY (receiver_wallet_id) REFERENCES Wallets(wallet_id) ON DELETE CASCADE );

CREATE TABLE AuditLog ( log_id INT AUTO_INCREMENT PRIMARY KEY,  
action_description VARCHAR(255) NOT NULL,  
log_timestamp TIMESTAMP DEFAULT CURRENT_TIMESTAMP );

Stored Procedure 
A stored procedure was implemented to handle the function of transferring funds. It encapsulates the 
transaction function (checking balances, locking rows, updating wallets) within the database layer for 
security and performance. 
DELIMITER $$  
CREATE PROCEDURE sp_TransferFunds(  
IN p_sender_wallet_id INT,  
IN p_receiver_wallet_id INT,  
IN p_amount DECIMAL(10, 2),  
OUT p_message VARCHAR(255) )  
BEGIN  
DECLARE v_sender_balance DECIMAL(10, 2);  
DECLARE EXIT HANDLER FOR SQLEXCEPTION BEGIN ROLLBACK;  
SET p_message = 'Transaction failed: Database error occurred.';  
END;   
IF NOT EXISTS (SELECT 1 FROM Wallets WHERE wallet_id = p_receiver_wallet_id) THEN SET 
p_message = 'Error: Receiver wallet does not exist.';  
ELSEIF p_sender_wallet_id = p_receiver_wallet_id THEN SET p_message = 'Error: Cannot transfer 
money to self.'; ELSE 
START TRANSACTION;  
SELECT balance INTO v_sender_balance FROM Wallets WHERE wallet_id = p_sender_wallet_id 
FOR UPDATE;  
IF v_sender_balance IS NULL THEN ROLLBACK;  
SET p_message = 'Error: Sender wallet not found.';  
ELSEIF v_sender_balance < p_amount THEN ROLLBACK; SET p_message = 'Error: Insufficient 
funds.';  
ELSE  
SET balance = balance - p_amount WHERE wallet_id = p_sender_wallet_id; 
UPDATE Wallets SET balance = balance + p_amount WHERE wallet_id = p_receiver_wallet_id;  
INSERT INTO Transactions (sender_wallet_id, receiver_wallet_id, amount, transaction_type) 
VALUES (p_sender_wallet_id, p_receiver_wallet_id, p_amount, 'TRANSFER');  
COMMIT; SET p_message = 'Transfer successful!';  
END IF;  
END IF;  
END$$ DELIMITER ;

Trigger 
A trigger was created on the Transactions table. Whenever a transaction occurs, this trigger automatically 
inserts a descriptive record into the AuditLog table for security monitoring. 
DELIMITER $$  
CREATE TRIGGER trg_AfterTransactionInsert  
AFTER INSERT ON Transactions  
FOR EACH ROW BEGIN  
DECLARE v_log_message VARCHAR(255);  
IF NEW.transaction_type = 'DEPOSIT' THEN SET v_log_message = CONCAT('New Deposit of $', 
NEW.amount, ' to Wallet ID ', NEW.receiver_wallet_id);

ELSEIF NEW.transaction_type = 'TRANSFER' THEN SET v_log_message = CONCAT('New Transfer

NEW.amount, ' from Wallet ID ', NEW.sender_wallet_id, ' to Wallet ID ', 
NEW.receiver_wallet_id);  
END IF;  
INSERT INTO AuditLog (action_description) VALUES (v_log_message);  
END$$ DELIMITER ;

View 
A SQL View was created to join the Transactions, Users, and Wallets tables. This simplifies complex 
queries when fetching the transaction history for the frontend. 
CREATE VIEW v_TransactionDetails AS SELECT t.transaction_id, t.transaction_date, t.amount, 
t.sender_wallet_id, sender_user.full_name AS sender_name,  
t.receiver_wallet_id, receiver_user.full_name AS receiver_name  
FROM Transactions t  
LEFT JOIN Wallets w_sender ON t.sender_wallet_id = w_sender.wallet_id  
LEFT JOIN Users sender_user ON w_sender.user_id = sender_user.user_id  
JOIN Wallets w_receiver ON t.receiver_wallet_id = w_receiver.wallet_id  
JOIN Users receiver_user ON w_receiver.user_id = receiver_user.user_id -- Filter: Show newest 
transactions first ORDER BY t.transaction_date DESC;
